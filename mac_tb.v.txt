module mac_tb;

    reg clk, rst, load;
    reg [3:0] A, B;
    wire [15:0] acc_out;

    mac_unit uut (
        .clk(clk),
        .rst(rst),
        .load(load),
        .A(A),
        .B(B),
        .acc_out(acc_out)
    );

    
    always #5 clk = ~clk;


    initial begin
        clk = 0;
        rst = 1;
        load = 0;
        A = 0;
        B = 0;

        #10 rst = 0;

        
        #10 load = 1; A = 4; B = 3;   // +12
        #10 A = 2; B = 5;             // +10
        #10 A = 1; B = 9;             // +9

        
        #10 load = 0; A = 7; B = 7;   // ignored

        #20 $finish;
    end

endmodule
