// Design Code

module mac_unit (
    input  wire        clk,
    input  wire        rst,      // synchronous reset
    input  wire        load,     // enable accumulation
    input  wire [3:0]  A,
    input  wire [3:0]  B,
    output reg  [15:0] acc_out
);

    wire [7:0] product;
    wire [15:0] product_ext;

    
    assign product = A * B;

    
    assign product_ext = {8'b0, product};

 
    always @(posedge clk) begin
        if (rst)
            acc_out <= 16'd0;
        else if (load)
            acc_out <= acc_out + product_ext;
    end

endmodule
